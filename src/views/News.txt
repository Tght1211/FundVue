<template>
    <div style="padding: 10px">
        <!-- 增加边距，上下10px，左右0px -->
        <!-- 功能区域 -->
        <div style="margin: 10px 0px">
            <el-button type="primary" @click="add">新增</el-button>
            <el-button type="primary">导入</el-button>
            <el-button type="primary">导出</el-button>
        </div>
        <!-- 搜索区域 -->
        <div style="margin: 10px 0px">
            <el-input v-model="search" placeholder="请输入书名" style="width: 20%" clearable/>
            <el-button type="primary" style="margin-left: 5px" @click="load">查询</el-button>
        </div>

        <!--    <img alt="Vue logo" src="../assets/logo.png">-->
        <el-table :data="tableData" style="width: 100%" >
            <el-table-column prop="id" label="ID" sortable/>
            <!-- mybatis-plus :后台数据库是下划线的话，前台会自动转为驼峰-->
            <el-table-column prop="title" label="标题"/>
            <el-table-column prop="author" label="作者"/>
            <el-table-column prop="time" label="日期"/>
            <el-table-column label="操作">
                <template #default="scope">
                    <!--   width="120" 当button加了这个，会出现溢出 -->
                    <el-button @click="details(scope.row)">详情</el-button>
                    <el-button @click="handleEdit(scope.row)">编辑</el-button>
                    <el-popconfirm title="你确定删除吗？" @confirm="handleDelete(scope.row.id)">
                        <template #reference>
                            <el-button type="danger">删除</el-button>
                        </template>
                    </el-popconfirm>
                </template>
            </el-table-column>
        </el-table>
        <div style="margin: 10px 0px">
            <el-pagination
                    v-model:currentPage="currentPage"
                    :page-sizes="[5, 10, 20]"
                    :page-size="pageSize"
                    :small="small"
                    :disabled="disabled"
                    :background="background"
                    layout="total, sizes, prev, pager, next, jumper"
                    :total="total"
                    @size-change="handleSizeChange"
                    @current-change="handleCurrentChange"
            >
            </el-pagination>

            <el-dialog
                    title="提示"
                    v-model="dialogVisible"
                    width="60%"
            >
                <el-form ref="form" :model="form" label-width="120px" :rules="rules">
                    <el-form-item label="标题" prop="title">
                        <el-input v-model="form.title" style="width: 30%"></el-input>
                    </el-form-item>
                  <!-- 内容 -->
                    <div id="div1" :ref="editor"></div>
                </el-form>
                <template #footer>
                    <span class="dialog-footer">
                        <el-button @click="dialogVisible = false">取 消</el-button>
                        <el-button type="primary" @click="save">确 定</el-button>
                    </span>
                </template>
            </el-dialog>

            <el-dialog
                    title="详情"
                    v-model="vis"
                    width="50%"
            >
                <h2 style="padding-left: 20px">{{detail.title}}</h2>
                <el-card style="min-height: 180px;margin-top: 20px ">
                    <div v-html="detail.content"></div>
                </el-card>
            </el-dialog>
        </div>
    </div>
</template>

<script>

    import request from "../utils/request";
    import router from "../router";
    import E from 'wangeditor'

    let editor
    export default {
        name: 'News',
        created() {
            this.load()
            const userJson = sessionStorage.getItem("user");
            if (!userJson) { // 不为空执行
                router.push("/login")
            }
        },
        mounted() {// 页面元素加载完后才会执行mounted代码

        },
        data() {
            return {
                editor:null,
                form: {},
                dialogVisible: false,
                search: '',
                currentPage: 1,
                pageSize: 10,
                total: 10,
                tableData: [],
                detail: {},
                vis: false,
                rules: {
                    title: [
                        {
                            required: true,
                            message: "请输入标题",
                            trigger: "blur",
                        },
                    ],
                }
            }
        },
        methods: {
            details(row){
                this.detail = JSON.parse(JSON.stringify(row))
                this.vis = true
            },
            filesUploadSuccess(res){
                console.log(res)
                this.form.cover = res.data
            },
            load() {
                request.get("/news/findPage", {
                    params: {
                        pageNum: this.currentPage,
                        pageSize: this.pageSize,
                        search: this.search
                    }
                }).then(res => {
                    console.log(res)
                    this.tableData = res.data.records
                    this.total = res.data.total
                })
            },
            add() {
                this.dialogVisible = true
                this.form = {}
              this.form.content = ''
                this.$nextTick(() => {
                    if (!editor){ //为空才可以
                        // 关联弹窗里面的div，new一个editor对象
                        editor = new E('#div1') // 打开弹窗，并且加载wangeditor这个对象
                        // 配置 server 接口地址
                        editor.config.uploadImgServer = 'http://' + window.server.filesUploadUrl + ':9090/files/editor'
                        editor.config.uploadFileName = "file" // 设置上传参数名称
                        editor.txt.html(this.form.content)
                        editor.create()
                    }else{
                        // 配置 server 接口地址
                        editor.txt.html(this.form.content)
                    }
                  this.form.content = ''
                })
            },
            save() {
               // console.log(editor.txt.html())
                this.form.content = editor.txt.html() // 获取编辑器里面的值，然后赋予到实体

                this.$refs['form'].validate((res) => {
                    if (res) {

                if (this.form.id) {
                    //更新
                    request.put("/news/update", this.form).then(res => {
                        console.log(res)
                        if (res.code === '0') {
                            this.$message({
                                type: "success",
                                message: "更新成功"
                            })
                        } else {
                            this.$message({
                                type: "error",
                                message: res.msg
                            })
                        }
                        this.dialogVisible = false
                        this.load()
                    })
                } else {
                    //新增
                    let userStr = sessionStorage.getItem("user")  || "{}"
                    let user = JSON.parse(userStr)
                    this.form.author = user.nickName

                    request.post("/news/save", this.form).then(res => {
                        console.log(res)
                        if (res.code === '0') {
                            this.$message({
                                type: "success",
                                message: "新增成功"
                            })
                        } else {
                            this.$message({
                                type: "error",
                                message: res.msg
                            })
                        }
                        this.dialogVisible = false
                        this.load()
                    })
                }
                    }
                })
            },
            handleEdit(row) {
                this.form = JSON.parse(JSON.stringify(row))
                this.dialogVisible = true

                this.$nextTick(() => {
                    if (!editor){ //为空才可以
                        // 关联弹窗里面的div，new一个editor对象
                        editor = new E('#div1') // 打开弹窗，并且加载wangeditor这个对象
                        // 配置 server 接口地址
                        editor.config.uploadImgServer = 'http://' + window.server.filesUploadUrl + ':9090/files/editor'
                        editor.config.uploadFileName = "file"// 设置上传参数名称
                        editor.txt.html(this.form.content)
                        editor.create()
                    }else{
                        // 配置 server 接口地址
                        editor.txt.html(this.form.content)
                    }
                })

            },
            handleDelete(id) {
                console.log(id)
                request.delete("/news/delete/" + id).then(res => {
                    if (res.code === '0') {
                        this.$message({
                            type: "success",
                            message: "删除成功"
                        })
                    } else {
                        this.$message({
                            type: "error",
                            message: res.msg
                        })
                    }
                    this.load()
                })
            },
            handleSizeChange(pageSize) { // 改变当前每页的个数触发
                this.pageSize = pageSize
                this.load()
            },
            handleCurrentChange(pageNum) { // 改变当前页码触发
                this.currentPage = pageNum
                this.load()
            }
        }
    }
</script>
